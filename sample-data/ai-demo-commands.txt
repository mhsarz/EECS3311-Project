# AI demo for the Smart Pantry CLI. Needs LLM_API_KEY in .env (uses about 15-20 AI calls).
# Run sample-data/setup-commands.txt first, then from the project folder:
#   java -jar target/recipe-agent-0.1.0-jar-with-dependencies.jar cli run sample-data/ai-demo-commands.txt

# F10 read the made-up sample flyers (the AI lists the sale items and prices)
flyers upload sample-data/flyers/sample-flyer-no-frills.txt --store "No Frills"
flyers upload sample-data/flyers/sample-flyer-metro.txt --store Metro
flyers upload sample-data/flyers/sample-flyer-freshco.html --store FreshCo
# A PDF flyer: page 1 has text, page 2 is only a picture of the prices; the AI reads both
flyers upload sample-data/flyers/sample-flyer-food-basics.pdf --store "Food Basics"
flyers list --search milk

# F02 sale alerts: milk and eggs are on sale; the cheapest store is picked
cart add milk --qty 4 --unit L
cart add eggs --qty 12
cart sales
cart list

# F01 recipe from pantry food, using what expires first (F05), saved to history (F08)
# (with nothing expiring soon, any pantry food is used)
recipe generate --time 30 --portions 2 --use-expiring --save

# F11 what is missing (an ingredient with another name in the pantry still counts)
recipe check last --add-missing

# F07 substitutes for one ingredient of the recipe: 2 = the second ingredient in the list
# (or give its name, e.g. recipe substitute last spinach; add --use 1 to put the first one in)
recipe substitute last 2

# F06 mark as cooked (takes the amounts out of the pantry), then undo
recipe cooked last
pantry list
undo
pantry list

# F08 history: rate, favourite, search
history rate last 5
history fav last
history list

# F09 chat with the agent (it can look at the pantry, cart, profile and flyers)
chat "What's on sale at No Frills that I could use this week?"
chat "Add a bag of carrots to my cart"
cart list
status
